# 预置提示词

```go
// MCP Prompt 示例：统一错误分析（alert_text 与 service_name 二选一）。
// 提示词正文来自同目录 error_analysis_prompt.md，运行时读取并渲染。
package main

import (
	"context"
	"embed"
	"fmt"
	"log"
	"net/http"
	"strings"
	"text/template"
	"time"

	"github.com/modelcontextprotocol/go-sdk/mcp"
)

//go:embed error_analysis_prompt.md
var promptFS embed.FS

var errorAnalysisTmpl = template.Must(
	template.New("error_analysis_prompt.md").ParseFS(promptFS, "error_analysis_prompt.md"),
)

type errorAnalysisPromptData struct {
	ServiceName string
	TimeRange   string
}

func promptHandler(_ context.Context, req *mcp.GetPromptRequest) (*mcp.GetPromptResult, error) {
	args := req.Params.Arguments
	svr := strings.TrimSpace(args["service_name"])
	timeRange := strings.TrimSpace(args["time_range"])

	data := errorAnalysisPromptData{
		ServiceName: svr,
		TimeRange:   timeRange,
	}

	var buf strings.Builder
	if err := errorAnalysisTmpl.Execute(&buf, data); err != nil {
		return nil, fmt.Errorf("渲染提示词模板: %w", err)
	}
	text := strings.TrimSpace(buf.String()) + "\n"
	return &mcp.GetPromptResult{
		Description: "错误分析：根据服务名和时间范围，自动查日志并分析根因",
		Messages: []*mcp.PromptMessage{
			{Role: "user", Content: &mcp.TextContent{Text: text}},
		},
	}, nil
}

func main() {
	s := mcp.NewServer(&mcp.Implementation{Name: "error-analysis-demo", Version: "0.1.0"}, nil)

	mcp.AddTool(s, &mcp.Tool{
		Name:        "list-service",
		Description: "列出所有服务",
	}, listServiceTool)

	mcp.AddTool(s, &mcp.Tool{
		Name:        "query-logs",
		Description: "查询日志",
		InputSchema: queryLogsInputSchema,
	}, queryLogsTool)

    s.AddPrompt(&mcp.Prompt{
		Name:        "error-analysis-prompt",
		Description: "错误分析：根据服务名和时间范围，自动查日志并分析根因",
		Arguments: []*mcp.PromptArgument{
			{Name: "service_name", Description: "服务名称"},
			{Name: "time_range", Description: `时间范围，如 "30m", "1h", 或 "start ~ end"`},
		},
	}, promptHandler)

	handler := mcp.NewStreamableHTTPHandler(func(*http.Request) *mcp.Server { return s }, nil)
	if err := http.ListenAndServe("127.0.0.1:8000", handler); err != nil {
		log.Fatal(err)
	}
}

type listServiceOutput struct {
	ServiceName string `json:"service_name" jsonschema:"服务名称"`
}

func listServiceTool(ctx context.Context, req *mcp.CallToolRequest, in any) (*mcp.CallToolResult, listServiceOutput, error) {
	return nil, listServiceOutput{
		ServiceName: "service1",
	}, nil
}

type queryLogsInput struct {
	ServiceName string   `json:"service_name" jsonschema:"服务名称"`
	LogLevels   []string `json:"log_levels,omitempty" jsonschema:"日志级别（可选，多选且不重复）；每项须为服务端枚举值之一"`
	TimeRange   string   `json:"time_range" jsonschema:"时间范围"`
}

type queryLogsOutput struct {
	Logs []string `json:"logs" jsonschema:"日志"`
}

func queryLogsTool(ctx context.Context, req *mcp.CallToolRequest, in queryLogsInput) (*mcp.CallToolResult, queryLogsOutput, error) {
	svc := strings.TrimSpace(in.ServiceName)
	if svc == "" {
		svc = "demo-service"
	}

	type fakeLine struct {
		level string
		at    time.Time
		msg   string
	}
	base := time.Date(2025, 3, 27, 10, 0, 0, 0, time.Local)
	// 10 条假数据：debug×2、info×4、warn×2、error×2（带时间与先后间隔）
	raw := []fakeLine{
		{"debug", base.Add(0 * time.Second), "配置热更新：拉取到 revision=4421"},
		{"debug", base.Add(3 * time.Second), "gRPC keepalive ping 正常"},
		{"info", base.Add(5 * time.Second), "HTTP GET /ready 200 cost=3ms"},
		{"info", base.Add(6 * time.Second), fmt.Sprintf("worker[%s] 消费偏移提交成功", svc)},
		{"info", base.Add(8 * time.Second), "异步任务 batch#88 入队"},
		{"info", base.Add(12 * time.Second), "metrics 上报周期完成"},
		{"warn", base.Add(45 * time.Second), "连接池 active=48/max=50"},
		{"warn", base.Add(46*time.Second + 200*time.Millisecond), "本地缓存淘汰 1200 条（LRU）"},
		{"error", base.Add(2 * time.Minute), "调用 billing.DecodeInvoice: context deadline exceeded"},
		{"error", base.Add(2*time.Minute + 3*time.Second), "MySQL 1213 deadlock, stmt=UPDATE quota SET ..."},
	}

	filter := make(map[string]struct{})
	for _, l := range in.LogLevels {
		k := strings.ToLower(strings.TrimSpace(l))
		if k != "" {
			filter[k] = struct{}{}
		}
	}

	out := make([]string, 0, len(raw))
	for _, e := range raw {
		if len(filter) > 0 {
			if _, ok := filter[e.level]; !ok {
				continue
			}
		}
		out = append(out, fmt.Sprintf("%s %s [%s] %s", e.at.Format("2006-01-02 15:04:05.000"), strings.ToUpper(e.level), svc, e.msg))
	}
	return nil, queryLogsOutput{Logs: out}, nil
}
```

## go:embed 编译指令

Go 1.16 起支持 `//go:embed` 编译指令，用来在编译时把文件（或目录）打进可执行文件里，而不是运行时再从磁盘读。  
`//go:embed` 必须是带 `//` 的注释，且紧挨着 import 以及被它修饰的变量。通常是 `embed.FS`、 `[]byte`、 `string`。

```go
//go:embed error_analysis_prompt.md
var promptFS embed.FS
```

不能嵌入会随构建变化、不可控的路径；通常用于模板、静态配置、提示词等固定资源。

## 提示词Handler

```go
func promptHandler(_ context.Context, req *mcp.GetPromptRequest) (*mcp.GetPromptResult, error) {
	// 获取输入的参数
	args := req.Params.Arguments
	svr := strings.TrimSpace(args["service_name"])
	timeRange := strings.TrimSpace(args["time_range"])

	// 构造提示词模板的数据
	data := errorAnalysisPromptData{
		ServiceName: svr,
		TimeRange:   timeRange,
	}

	// 渲染提示词模板
	var buf strings.Builder
	if err := errorAnalysisTmpl.Execute(&buf, data); err != nil {
		return nil, fmt.Errorf("渲染提示词模板: %w", err)
	}
	text := strings.TrimSpace(buf.String()) + "\n"

	// 返回提示词结果
	return &mcp.GetPromptResult{
		Description: "错误分析：根据服务名和时间范围，自动查日志并分析根因",
		Messages: []*mcp.PromptMessage{
			{Role: "user", Content: &mcp.TextContent{Text: text}},
		},
	}, nil
}
```

| 字段名 | 作用 |
| --- | --- |
| Role | 说话方角色，协议上为 `user` 或 `assistant`。 |
| Content | 单条消息的内容块；类型由 JSON 里 `type` 区分，可包含内嵌资源等多形态内容。 |

Content 不仅仅在Prompt有用到，它是一个公共定义。具体内容详见[Content](Content.md)。

上边例子使用的是mcp.TextContent， 是一个文本内容。我们还可以在返回的内容追加一张图片，例如一个流程图。

![error_analysis_flowchart](assets/1774693103282-bd7c805e-5590-4637-9b7a-cffefb4a6d1d.png)

```go
//go:embed error_analysis_flowchart.png
var errorAnalysisFlowchartPNG []byte
```

```go
	// 返回提示词结果
	return &mcp.GetPromptResult{
		Description: "错误分析：根据服务名和时间范围，自动查日志并分析根因",
		Messages: []*mcp.PromptMessage{
			{Role: "user", Content: &mcp.TextContent{Text: text}},

				Role: "user",

					MIMEType: "image/png",
					Data:     errorAnalysisFlowchartPNG,
				},
			},
		},
	}, nil
```

![image](assets/1774693257493-958d5464-da1e-47f5-acf5-725cbe91762d.png)

`mcp.ImageContent.MIMEType`即**MIME**，最初指 *Multipurpose Internet Mail Extensions*（多用途互联网邮件扩展），用来在邮件里标明附件是图片、音频还是别的格式。如今更广为人知的名字是 **媒体类型（Media Type）** 或 **内容类型（Content Type）**。

取值是写死的字符串，详见 [MIME 类型](../%E5%8F%82%E8%80%83/MIME%20%E7%B1%BB%E5%9E%8B.md)
